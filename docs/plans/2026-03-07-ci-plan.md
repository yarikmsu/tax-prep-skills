### Task 1: Create the GitHub Actions CI workflow

**Files:**
- Create: `.github/workflows/ci.yml`

**Step 1: Create the directory structure**

Run:
```bash
mkdir -p .github/workflows
```

**Step 2: Write the workflow file**

```yaml
name: CI

on:
  pull_request:
    branches: [main]

jobs:
  test:
    name: Tests
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - name: Install dependencies
        run: pip install -r requirements.txt
      - name: Install test dependencies
        run: pip install pytest
      - name: Run tests
        run: python -m pytest tests/ -v

  pii-scan:
    name: PII file scan
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Check for blocked file types
        run: |
          BLOCKED_EXTENSIONS="pdf|xlsx|xls|csv|doc|docx|png|jpg|jpeg|gif|bmp|tiff|heic|zip|tar|gz"
          VIOLATIONS=""

          for file in $(git diff --name-only origin/main...HEAD 2>/dev/null || git ls-files); do
            filename=$(basename "$file")

            # Check blocked extensions
            if echo "$filename" | grep -qiE "\.($BLOCKED_EXTENSIONS)$"; then
              VIOLATIONS="${VIOLATIONS}  BLOCKED FILE TYPE: $file\n"
              continue
            fi

            # Check blocked filenames
            if [ "$filename" = "taxpayer_profile.json" ] || [ "$filename" = "result.json" ]; then
              VIOLATIONS="${VIOLATIONS}  BLOCKED FILENAME: $file\n"
              continue
            fi

            # Check tax_law_*.json pattern
            if echo "$filename" | grep -qE '^tax_law_.*\.json$'; then
              VIOLATIONS="${VIOLATIONS}  BLOCKED FILENAME: $file\n"
              continue
            fi
          done

          if [ -n "$VIOLATIONS" ]; then
            echo ""
            echo "=========================================="
            echo "  CI BLOCKED — Sensitive files found"
            echo "=========================================="
            echo ""
            echo -e "$VIOLATIONS"
            exit 1
          fi

          echo "No blocked file types found."

  security:
    name: Dependency audit
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - name: Install pip-audit
        run: pip install pip-audit
      - name: Audit dependencies
        run: pip-audit -r requirements.txt
```
